  public class QandA
  {
    public string Question { get; set; }
    public string Answer { get; set; }
    public bool Answered;
    public bool Correct;
    public static int Points { get; set; }

    public QandA(string question, string answer)
    {
      Question = question;
      Answer = answer;
    }
    public string CheckAnswer(string userInput)
    {
        if (userInput.Trim().ToLower() == Answer){
            Answered = true;
            Correct = true;
            Points++;
            return "Riktig!";
        }
        else if (userInput != string.Empty) {
            Answered = true;
            return "Feil.";
        }
        else {
            return "Ikke besvart.";
        }
    }
    public string CheckAnswerByBool()
    {
        if (Answered && Correct) return "Riktig!";
        else if (!Answered) return "Ikke besvart.";
        else return "Feil.";
    }
  }
